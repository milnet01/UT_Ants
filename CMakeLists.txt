cmake_minimum_required(VERSION 3.28)

project(UT_Ants
    VERSION 0.0.1
    DESCRIPTION "A from-scratch engine that bakes Unreal Tournament maps into modernised versions of themselves"
    LANGUAGES CXX)

# C++23 rather than cpp.md's C++20 floor. The reason is std::expected, which
# docs/design.md makes the error model across every module boundary; without
# it the floor would buy a hand-rolled equivalent in core and nothing else.
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# This project uses no C++20 modules. Left on, CMake writes GCC's module
# scanning flags -- -fmodules-ts, -fmodule-mapper=, -fdeps-format= -- into
# compile_commands.json, and no clang-based tool can parse them: clang-tidy
# reports them as errors and an editor's clangd fails to index the file at
# all. Turning it off costs nothing here and makes the compilation database
# usable by the tools that read it.
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)

# docs/design.md § The stack rules out anything older. Checked here rather
# than left to a confusing template error four thousand lines into a build.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 14)
    message(FATAL_ERROR "GCC 14 or newer is required (found ${CMAKE_CXX_COMPILER_VERSION}); see docs/design.md")
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 19)
    # 19, not 18: Clang 18 reports __cpp_concepts as 201907L and libstdc++
    # gates std::expected on 202002L, so <expected> compiles away entirely.
    # Measured in an ubuntu:24.04 container on 2026-09-04 against both.
    message(FATAL_ERROR "Clang 19 or newer is required (found ${CMAKE_CXX_COMPILER_VERSION}); see docs/design.md")
elseif(MSVC AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 19.40)
    message(FATAL_ERROR "MSVC 19.40 / Visual Studio 2022 17.10 or newer is required (found ${CMAKE_CXX_COMPILER_VERSION}); see docs/design.md")
endif()

# Windows is a first-class target (docs/design.md § The stack), so MSVC's two
# defaults that differ from every other compiler here are set once, globally.
if(MSVC)
    # Without /utf-8 MSVC reads sources in the machine's ANSI code page and the
    # non-ASCII in this project's own comments becomes a warning storm or a
    # mojibake string literal. It is not a style choice: the other two
    # compilers assume UTF-8 and this one has to be told.
    add_compile_options(/utf-8)
    # The traditional preprocessor is not conforming and breaks variadic macros
    # the same way for every library that uses them, Catch2 included.
    add_compile_options(/Zc:preprocessor)
endif()

# docs/design.md § "What every part does the same way", the Determinism
# bullet: floating-point contraction and fast-math off on every compiler,
# because two first-class toolchains do not round the same way by default
# and ADR-0002 needs one recipe and baker version to hash to one bundle on
# any machine. docs/specs/UTA-0049-numeric-contract.md is the contract this
# enforces and tests/unit/NumericContractTest.cpp is what proves it holds.
#
# Measured 2026-09-04 (GCC 16.2.0, Clang 22.1.8, this machine): with an FMA
# target (-march=native) both compilers fold `a*b+c` into a single `vfmadd`
# with NO flag at all, and -ffp-contract=off is what suppresses it -- so
# contraction is the one part of this contract that is NOT already the
# compiler's default and needs a flag. The rest of the -ffast-math bundle
# (-fassociative-math, -ffinite-math-only, -freciprocal-math, signed zeros,
# trapping math, math errno) is already off/on correctly with no flag
# passed at all (checked via `gcc -Q --help=optimizers`); -fno-fast-math is
# set anyway, explicit rather than relied on, matching this file's own
# style of not trusting a compiler default silently -- and because it is
# what actually reverts -ffast-math (or the deprecated -Ofast) if one
# arrives from CMAKE_CXX_FLAGS ahead of these, verified by disassembly.
if(MSVC)
    # /fp:precise, and only that. Microsoft's /fp reference gives the
    # option-to-pragma mapping as /fp:precise -> fp_contract(off) from
    # Visual Studio 2022 on, and says in prose: "Floating-point
    # contractions aren't generated by default under /fp:precise. This
    # behavior is new in Visual Studio 2022." This project's floor is
    # MSVC 19.40 / VS 2022 17.10, checked above, so that is the branch
    # that applies. /fp:precise is also MSVC's default; it is set
    # explicitly rather than trusted, for the same reason as /utf-8 and
    # /Zc:preprocessor above, and so that the refuse guard below is not
    # the only thing standing between this build and /fp:fast.
    #
    # There is deliberately no second flag here. /fp:contract is the
    # opt-IN, and the reference's syntax line -- "/fp:contract
    # /fp:except[-] /fp:fast /fp:precise /fp:strict" -- gives a trailing
    # minus to /fp:except alone, so /fp:contract- is not an option that
    # exists. An earlier draft of this block passed it.
    #
    # UNVERIFIED BY COMPILATION: this machine has no MSVC, and this
    # project's Windows box runs binaries but cannot build. The Windows CI
    # leg is what gates this flag, not this comment.
    add_compile_options(/fp:precise)
else()
    # GCC and Clang, spelling verified by disassembling a throwaway
    # translation unit on this machine (docs/specs/UTA-0049-numeric-contract.md § 2).
    add_compile_options(-ffp-contract=off -fno-fast-math)
endif()

# Refuse rather than silently lose: if CMAKE_CXX_FLAGS (or a per-config
# equivalent) carries a flag that would turn contraction or fast-math back
# on, configuration fails loudly instead of proceeding with a build that
# silently disagrees with what was asked for. The flags above are appended
# after CMAKE_CXX_FLAGS in the actual compiler invocation on the generators
# this project uses (verified against a throwaway CMake+Ninja project,
# 2026-09-04), so they would usually win by ordering alone -- but that
# ordering is not part of any interface contract, and a build that quietly
# discards a flag someone deliberately passed is a worse failure than one
# that refuses to configure: the person who passed it does not find out it
# did nothing until the numbers stop matching across machines, which is
# exactly the failure this whole contract exists to prevent. Same idiom as
# the UTA_SANITIZE and compiler-version checks below: explain why, then
# fail fast.
set(_uta_unsafe_fp_pattern "(-ffast-math|-Ofast|/fp:fast|-ffp-contract=fast)")
foreach(_uta_fp_flags_var IN ITEMS
        CMAKE_CXX_FLAGS CMAKE_CXX_FLAGS_DEBUG CMAKE_CXX_FLAGS_RELEASE
        CMAKE_CXX_FLAGS_RELWITHDEBINFO CMAKE_CXX_FLAGS_MINSIZEREL)
    if(${_uta_fp_flags_var} MATCHES "${_uta_unsafe_fp_pattern}")
        message(FATAL_ERROR
            "${_uta_fp_flags_var} ('${${_uta_fp_flags_var}}') passes a "
            "fast-math or contraction-enabling flag. docs/design.md's "
            "Determinism bullet requires both off on every toolchain -- "
            "ADR-0002 needs one recipe and baker version to hash to one "
            "bundle on any machine, and that breaks the moment one build "
            "enables an optimisation the standard does not require to "
            "round the same way. Remove the flag rather than passing it "
            "through ${_uta_fp_flags_var}.")
    endif()
endforeach()
# _uta_unsafe_fp_pattern is deliberately NOT unset: src/umat/CMakeLists.txt
# asserts INV-8 of docs/specs/UTA-0052-texture-memory-budget.md with this same
# pattern, and reading it is what keeps the two checks from drifting apart.
unset(_uta_fp_flags_var)

# Build speed. Both tools are OPTIONAL and detected: a machine without either
# produces the same binaries, just slower, so a clone still builds and tests
# clean and S7 is untouched. Measured on ROADMAP UTA-0050, which carries the
# figures and their date rather than repeating them here.
#
# ccache is the one that pays on this project. Nearly every object in a cold
# build is Catch2, which never changes, so a wiped build directory recompiles
# the test framework for nothing; with a warm cache a fresh build directory
# is close to instant. It needs two settings CMake cannot make for you --
# ccache's defaults hash the build directory into the key, which scores badly
# across two of them:
#
#     ccache --set-config base_dir=/
#     ccache --set-config hash_dir=false
#
# The flags above are part of the compile line, so the cache key changes with
# them: a build that alters the numeric contract cannot hit an entry compiled
# under the old one.
find_program(UTA_CCACHE ccache)
if(UTA_CCACHE)
    set(CMAKE_CXX_COMPILER_LAUNCHER "${UTA_CCACHE}")
    message(STATUS "Compiler launcher: ${UTA_CCACHE}")
endif()

# mold saves almost nothing TODAY -- this project links one static library and
# one test executable, so there is barely a link step to speed up. It is wired
# now because that stops being true once urender and the third-party
# dependencies arrive, and a detected flag costs nothing in the meantime.
# Not on MSVC, which has no -fuse-ld.
if(NOT MSVC)
    find_program(UTA_MOLD mold)
    if(UTA_MOLD)
        add_link_options(-fuse-ld=mold)
        message(STATUS "Linker: mold (${UTA_MOLD})")
    endif()
endif()

# The job system is the first threaded code in this project, and ordinary
# test runs do not see data races. This is what INV-13 of
# docs/specs/UTA-0002-core-foundations.md is measured with.
#
# `thread` and nothing else. AddressSanitizer is not offered here because MSVC
# spells it /fsanitize=address and takes no link flag, so a second value would
# need a second spelling for a build nothing in this repository runs.
set(UTA_SANITIZE "" CACHE STRING "Sanitizer to build with: empty or thread")
set_property(CACHE UTA_SANITIZE PROPERTY STRINGS "" "thread")

if(UTA_SANITIZE STREQUAL "thread")
    if(MSVC)
        message(FATAL_ERROR
            "UTA_SANITIZE=thread is not available on MSVC, which has no "
            "ThreadSanitizer. Refusing rather than building something that "
            "does not check what it claims.")
    endif()
    # Applied at the TOP LEVEL on purpose: uta_core, the test executable and
    # the Catch2 the build fetches must all be instrumented. Instrumenting the
    # library alone yields a leg that reports nothing about the job tests,
    # which is what INV-13 is actually about.
    add_compile_options(-fsanitize=thread -g -O1)
    add_link_options(-fsanitize=thread)
elseif(NOT UTA_SANITIZE STREQUAL "")
    message(FATAL_ERROR
        "UTA_SANITIZE is '${UTA_SANITIZE}'; the accepted values are '' and 'thread'.")
endif()

option(UTA_BUILD_TESTS "Build the test suite" ON)

# The command-line tools. ON by default: they link only libraries this
# build already produces, so a clone with no Unreal Tournament builds them
# exactly as it builds everything else, which is what S7 is measured on.
option(UTA_BUILD_TOOLS "Build the command-line tools" ON)

# The second test tier. OFF by default so a clone with no Unreal Tournament
# builds and tests clean, which is what S7 is measured on. Turning it on
# without a path is a configure error rather than a silently empty run.
option(UTA_REAL_ASSET_TESTS "Also run tests against a real Unreal Tournament install" OFF)
set(UTA_UT_INSTALL_DIR "" CACHE PATH "Unreal Tournament install to run UTA_REAL_ASSET_TESTS against")

if(UTA_REAL_ASSET_TESTS AND NOT UTA_UT_INSTALL_DIR)
    message(FATAL_ERROR
        "UTA_REAL_ASSET_TESTS is ON but UTA_UT_INSTALL_DIR is empty. "
        "Point it at an Unreal Tournament install, or leave both at their defaults.")
endif()

add_subdirectory(src)

if(UTA_BUILD_TOOLS)
    add_subdirectory(tools)
endif()

if(UTA_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
